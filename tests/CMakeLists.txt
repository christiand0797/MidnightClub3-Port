cmake_minimum_required(VERSION 3.25)
project(MC3RuntimeTests LANGUAGES CXX)
enable_testing()
foreach(test voice3d winsock_version shader_stage input_mapping timing tuning modulation garage camera_lod)
    add_executable(${test}_test ${test}_test.cpp)
    target_include_directories(${test}_test PRIVATE ../runtime)
    target_compile_features(${test}_test PRIVATE cxx_std_20)
    # Keep runtime checks enabled in Release as well.
    if(MSVC)
        target_compile_options(${test}_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(${test}_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
target_sources(shader_stage_test PRIVATE ../runtime/gpu/shaders.cpp)
target_sources(timing_test PRIVATE ../templates/game/src/game/hooks.cpp)
target_include_directories(timing_test PRIVATE ../templates/game/generated/game)
target_sources(tuning_test PRIVATE ../templates/game/src/game/tuning.cpp)
target_include_directories(tuning_test PRIVATE ../templates/game/generated/game)
target_sources(garage_test PRIVATE ../templates/game/src/game/garage.cpp)
target_include_directories(garage_test PRIVATE ../templates/game/generated/game)
target_sources(camera_lod_test PRIVATE ../templates/game/src/game/lod.cpp ../templates/game/src/game/fov.cpp)
target_include_directories(camera_lod_test PRIVATE ../templates/game/generated/game)

if(WIN32)

set(MC3_HD_TEST_ROOT "" CACHE PATH "Directory containing video_hd/rockstar.mp4 for Media Foundation testing")

    target_include_directories(video_hd_test PRIVATE ../runtime)
    target_compile_features(video_hd_test PRIVATE cxx_std_20)
    target_link_libraries(video_hd_test PRIVATE mfplat mfreadwrite mfuuid ole32)
    if(MSVC)
        target_compile_options(video_hd_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(video_hd_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME video_hd COMMAND video_hd_test "${MC3_HD_TEST_ROOT}")
    set_tests_properties(video_hd PROPERTIES TIMEOUT 45)
endif()
